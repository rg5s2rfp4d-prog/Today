    const STORAGE_KEY = 'aether_glass_life_os_v1';

    function formatDateKey(dateObj) {
      const y = dateObj.getFullYear();
      const m = String(dateObj.getMonth() + 1).padStart(2, '0');
      const d = String(dateObj.getDate()).padStart(2, '0');
      return `${y}-${m}-${d}`;
    }

    function parseDateKey(dateStr) {
      const [y, m, d] = dateStr.split('-').map(Number);
      return new Date(y, m - 1, d);
    }

    const TODAY_KEY = formatDateKey(new Date());

    // Application State (Defaulting to Obsidian Dark Monochrome for Leica / iOS 26 aesthetic)
    let state = {
      lang: 'zh',       // 'zh' | 'en'
      theme: 'dark',    // 'dark' (Obsidian Black) | 'light' (Pure Frost White)
      activeView: 'today',
      selectedDate: TODAY_KEY,
      calendarMonth: TODAY_KEY.slice(0, 7),
      selectedYear: new Date().getFullYear(),

      // User Data Entities
      tasks: [],        // { id, title, date, note, completed, priority, createdAt }
      habits: [],       // { id, title, titleEn, icon, createdAt }
      habitLogs: {},    // { 'YYYY-MM-DD': ['hab_id_1', ...] }
      journals: {},     // { 'YYYY-MM-DD': { content, updatedAt } }
      moodLogs: {},     // { 'YYYY-MM-DD': 'mood_id' }
      customMoods: [],  // { id, emoji, zh, en }
      annualGoals: []   // { id, year, title, category, progress, completed, createdAt }
    };

    /**
     * Non-destructive migration & bidirectional compatibility with localStorage #41777
     * Preserves every existing task, habit, journal entry, mood, and goal.
     */
    function loadAndMigrateStorage() {
      let existingMain = null;
      try {
        const rawMain = localStorage.getItem(STORAGE_KEY);
        if (rawMain) existingMain = JSON.parse(rawMain);
      } catch (e) {
        console.warn('Error reading main storage:', e);
      }

      if (existingMain && typeof existingMain === 'object') {
        state = {
          ...state,
          ...existingMain,
          tasks: Array.isArray(existingMain.tasks) ? existingMain.tasks : [],
          habits: Array.isArray(existingMain.habits) ? existingMain.habits : [],
          habitLogs: existingMain.habitLogs && typeof existingMain.habitLogs === 'object' ? existingMain.habitLogs : {},
          journals: existingMain.journals && typeof existingMain.journals === 'object' ? existingMain.journals : {},
          moodLogs: existingMain.moodLogs && typeof existingMain.moodLogs === 'object' ? existingMain.moodLogs : {},
          customMoods: Array.isArray(existingMain.customMoods) ? existingMain.customMoods : [],
          annualGoals: Array.isArray(existingMain.annualGoals) ? existingMain.annualGoals : []
        };
      }

      // Scan all localStorage keys for legacy data (#41777, tasks, todos, habits, journal, moods, etc.)
      const legacySnapshot = {};
      let migratedCount = 0;

      for (let i = 0; i < localStorage.length; i++) {
        const key = localStorage.key(i);
        if (!key || key === STORAGE_KEY || key === LEGACY_BACKUP_KEY) continue;

        const rawVal = localStorage.getItem(key);
        if (!rawVal) continue;
        legacySnapshot[key] = rawVal;

        try {
          const parsed = JSON.parse(rawVal);
          const lowerKey = key.toLowerCase();

          // 1. Migrate legacy tasks / todos arrays
          if ((lowerKey.includes('task') || lowerKey.includes('todo') || lowerKey.includes('41777')) && Array.isArray(parsed)) {
            parsed.forEach((item, idx) => {
              if (!item) return;
              const title = typeof item === 'string' ? item : (item.title || item.text || item.name || item.content);
              if (!title) return;
              const id = item.id ? String(item.id) : `legacy_task_${key}_${idx}_${title.slice(0, 8)}`;
              if (!state.tasks.some(t => String(t.id) === id || (t.title === title && t.date === (item.date || TODAY_KEY)))) {
                state.tasks.push({
                  id,
                  title: String(title),
                  note: item.note || item.desc || '',
                  date: item.date || item.dueDate || TODAY_KEY,
                  completed: Boolean(item.completed || item.done || item.isCompleted),
                  priority: Boolean(item.priority || item.important || item.starred || item.isPriority),
                  createdAt: item.createdAt || new Date().toISOString()
                });
                migratedCount++;
              }
            });
          }

          // 2. Migrate composite planner object
          if (parsed && typeof parsed === 'object' && !Array.isArray(parsed)) {
            if (Array.isArray(parsed.tasks)) {
              parsed.tasks.forEach((t, idx) => {
                if (t && (t.title || t.text)) {
                  const id = String(t.id || `legacy_obj_task_${idx}`);
                  if (!state.tasks.some(x => String(x.id) === id)) {
                    state.tasks.push({
                      id,
                      title: t.title || t.text,
                      note: t.note || '',
                      date: t.date || TODAY_KEY,
                      completed: Boolean(t.completed || t.done),
                      priority: Boolean(t.priority || t.important),
                      createdAt: t.createdAt || new Date().toISOString()
                    });
                    migratedCount++;
                  }
                }
              });
            }
            if (Array.isArray(parsed.habits)) {
              parsed.habits.forEach((h, idx) => {
                if (h && (h.title || h.name)) {
                  const id = String(h.id || `legacy_hab_${idx}`);
                  if (!state.habits.some(x => String(x.id) === id)) {
                    state.habits.push({
                      id,
                      title: h.title || h.name,
                      titleEn: h.titleEn || h.title || h.name,
                      icon: h.icon || h.emoji || '◇',
                      createdAt: h.createdAt || TODAY_KEY
                    });
                    migratedCount++;
                  }
                }
              });
            }
            if (parsed.journals && typeof parsed.journals === 'object') {
              Object.entries(parsed.journals).forEach(([dKey, jVal]) => {
                if (!state.journals[dKey] && jVal) {
                  state.journals[dKey] = typeof jVal === 'string'
                    ? { content: jVal, updatedAt: new Date().toISOString() }
                    : jVal;
                  migratedCount++;
                }
              });
            }
            if (parsed.moods && typeof parsed.moods === 'object' && !Array.isArray(parsed.moods)) {
              Object.entries(parsed.moods).forEach(([dKey, mVal]) => {
                if (!state.moodLogs[dKey] && mVal) {
                  state.moodLogs[dKey] = mVal;
                  migratedCount++;
                }
              });
            }
            if (Array.isArray(parsed.goals || parsed.annualGoals)) {
              (parsed.goals || parsed.annualGoals).forEach((g, idx) => {
                if (g && g.title) {
                  const id = String(g.id || `legacy_goal_${idx}`);
                  if (!state.annualGoals.some(x => String(x.id) === id)) {
                    state.annualGoals.push({
                      id,
                      year: g.year || new Date().getFullYear(),
                      title: g.title,
                      category: g.category || 'Personal',
                      progress: Number(g.progress || 0),
                      completed: Boolean(g.completed),
                      createdAt: g.createdAt || new Date().toISOString()
                    });
                    migratedCount++;
                  }
                }
              });
            }
          }

          // 3. Migrate standalone habits array
          if (lowerKey.includes('habit') && Array.isArray(parsed)) {
            parsed.forEach((h, idx) => {
              const title = typeof h === 'string' ? h : (h.title || h.name);
              if (!title) return;
              const id = String(h.id || `legacy_habit_${idx}`);
              if (!state.habits.some(x => String(x.id) === id || x.title === title)) {
                state.habits.push({
                  id,
                  title,
                  titleEn: h.titleEn || title,
                  icon: h.icon || h.emoji || '◇',
                  createdAt: TODAY_KEY
                });
                migratedCount++;
              }
            });
          }

          // 4. Migrate standalone journal map
          if (lowerKey.includes('journal') && parsed && typeof parsed === 'object' && !Array.isArray(parsed)) {
            Object.entries(parsed).forEach(([dKey, entry]) => {
              if (!state.journals[dKey] && entry) {
                state.journals[dKey] = typeof entry === 'string'
                  ? { content: entry, updatedAt: new Date().toISOString() }
                  : entry;
                migratedCount++;
              }
            });
          }
        } catch (err) {
          if (key.toLowerCase().includes('journal') && rawVal.trim().length > 0 && !state.journals[TODAY_KEY]) {
            state.journals[TODAY_KEY] = { content: rawVal, updatedAt: new Date().toISOString() };
            migratedCount++;
          }
        }
      }

      if (Object.keys(legacySnapshot).length > 0 && !localStorage.getItem(LEGACY_BACKUP_KEY)) {
        try {
          localStorage.setItem(LEGACY_BACKUP_KEY, JSON.stringify(legacySnapshot));
        } catch (e) {}
      }

      // Seed minimalist starter items ONLY if completely empty (zero existing user data)
      if (!existingMain && migratedCount === 0 && state.tasks.length === 0 && state.habits.length === 0) {
        state.habits = [...DEFAULT_HABITS];
        state.tasks = [
          {
            id: 'welcome_1',
            title: '聚焦今日最關鍵的三項核心任務',
            note: '保持純粹與專注，逐一完成',
            date: TODAY_KEY,
            completed: false,
            priority: true,
            createdAt: new Date().toISOString()
          },
          {
            id: 'welcome_2',
            title: '整理晚間思緒與隨手日記',
            note: '',
            date: TODAY_KEY,
            completed: false,
            priority: false,
            createdAt: new Date().toISOString()
          }
        ];
        state.annualGoals = [
          {
            id: 'goal_demo_1',
            year: new Date().getFullYear(),
            title: '建立極簡高專注的日常創作與閱讀節奏',
            category: '核心秩序',
            progress: 40,
            completed: false,
            createdAt: new Date().toISOString()
          }
        ];
      }

      // Always land on Today Home on initial entry
      state.selectedDate = TODAY_KEY;
      state.calendarMonth = TODAY_KEY.slice(0, 7);
      state.activeView = 'today';

      saveState();
    }

      }
    }

    function formatHumanDate(dateStr, lang = state.lang) {
      const d = parseDateKey(dateStr);
      if (lang === 'zh') {
        const weekdays = ['星期日', '星期一', '星期二', '星期三', '星期四', '星期五', '星期六'];
        return `${d.getFullYear()}年${d.getMonth() + 1}月${d.getDate()}日 · ${weekdays[d.getDay()]}`;
      } else {
        return d.toLocaleDateString('en-US', {
          weekday: 'short',
          year: 'numeric',
          month: 'short',
          day: 'numeric'
        });
      }
    }

    function calculateHabitStreak(habitId) {
      let streak = 0;
      const cursor = parseDateKey(TODAY_KEY);
      if (!(state.habitLogs[TODAY_KEY] || []).includes(habitId)) {
        cursor.setDate(cursor.getDate() - 1);
      }
      while (true) {
        const key = formatDateKey(cursor);
        if ((state.habitLogs[key] || []).includes(habitId)) {
          streak++;
          cursor.setDate(cursor.getDate() - 1);
        } else {
          break;
        }
      }
      return streak;
    }

    function exportDataJson() {
      const blob = new Blob([JSON.stringify(state, null, 2)], { type: 'application/json' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `mono-glass-backup-${TODAY_KEY}.json`;
      a.click();
      URL.revokeObjectURL(url);
      showToast(state.lang === 'zh' ? '已下載備份檔案' : 'Backup downloaded');
    }

    function importDataJson(event) {
      const file = event.target.files && event.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = (e) => {
        try {
          const imported = JSON.parse(e.target.result);
          if (imported && typeof imported === 'object') {
            state = {
              ...state,
              ...imported,
              tasks: Array.isArray(imported.tasks) ? imported.tasks : state.tasks,
              habits: Array.isArray(imported.habits) ? imported.habits : state.habits,
              habitLogs: imported.habitLogs || state.habitLogs,
              journals: imported.journals || state.journals,
              moodLogs: imported.moodLogs || state.moodLogs,
              customMoods: Array.isArray(imported.customMoods) ? imported.customMoods : state.customMoods,
              annualGoals: Array.isArray(imported.annualGoals) ? imported.annualGoals : state.annualGoals
            };
            saveState();
            closeModal();
            renderApp();
            showToast(state.lang === 'zh' ? '資料已成功還原' : 'Data restored');
          }
        } catch (err) {
          showToast(state.lang === 'zh' ? '檔案格式無效' : 'Invalid JSON file');
        }
      };
      reader.readAsText(file);
    }

    function escapeHtml(str) {
      return String(str || '')
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;');
    }

    function renderWeekView() {
      const anchor = parseDateKey(state.selectedDate);
      const dayOfWeek = anchor.getDay();
      const mondayOffset = dayOfWeek === 0 ? -6 : 1 - dayOfWeek;
      const monday = new Date(anchor);
      monday.setDate(anchor.getDate() + mondayOffset);

      const days = [];
      let weekTasksTotal = 0;
      let weekTasksDone = 0;
      let weekHabitsDone = 0;

      for (let i = 0; i < 7; i++) {
        const d = new Date(monday);
        d.setDate(monday.getDate() + i);
        const key = formatDateKey(d);
        const tasks = state.tasks.filter(t => t.date === key);
        const done = tasks.filter(t => t.completed).length;
        const habitsDone = (state.habitLogs[key] || []).length;
        const moodId = state.moodLogs[key];
        const moodObj = getAllMoods().find(m => m.id === moodId);
        const journal = state.journals[key];

        weekTasksTotal += tasks.length;
        weekTasksDone += done;
        weekHabitsDone += habitsDone;

        days.push({ date: d, key, tasks, done, habitsDone, moodObj, journal });
      }

      const weekdayNames = state.lang === 'zh'
        ? ['週一', '週二', '週三', '週四', '週五', '週六', '週日']
        : ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];

      return `
        <!-- Week Summary Strip -->
        <section class="liquid-glass rounded-3xl p-4 mb-4 flex items-center justify-between gap-3">
          <div>
            <h2 class="text-sm font-bold">${t('weekSummary')}</h2>
            <p class="text-xs opacity-60 font-mono">${days[0].key} — ${days[6].key}</p>
          </div>
          <div class="flex items-center gap-5 text-right">
            <div>
              <div class="text-lg font-bold editorial-num">${weekTasksDone}/${weekTasksTotal}</div>
              <div class="text-[10px] opacity-65">${t('tasksDoneLabel')}</div>
            </div>
            <div>
              <div class="text-lg font-bold editorial-num">${weekHabitsDone}</div>
              <div class="text-[10px] opacity-65">${t('habitsDoneLabel')}</div>
            </div>
          </div>
        </section>

        <!-- 7-Day Interactive Liquid Glass Cards -->
        <div class="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">
          ${days.map((item, idx) => {
            const isToday = item.key === TODAY_KEY;
            const isSelected = item.key === state.selectedDate;
            return `
              <div onclick="selectDateAndOpenToday('${item.key}')"
                class="liquid-glass rounded-3xl p-4 cursor-pointer hover:scale-[1.01] transition ${isSelected ? 'ring-1.5 ring-current' : ''}">
                <div class="flex items-center justify-between mb-2.5">
                  <div class="flex items-center gap-2">
                    <span class="px-2.5 py-1 rounded-xl text-xs font-bold ${isToday ? 'mono-solid-btn' : 'mono-badge'}">
                      ${weekdayNames[idx]} · ${item.date.getMonth() + 1}/${item.date.getDate()}
                    </span>
                    ${item.moodObj ? `<span class="text-sm font-bold mono-symbol" title="${escapeHtml(item.moodObj.zh)}">${escapeHtml(item.moodObj.emoji)}</span>` : ''}
                  </div>
                  <span class="text-xs font-mono opacity-65">
                    ${item.done}/${item.tasks.length}
                  </span>
                </div>

                <!-- Tasks Preview -->
                <div class="space-y-1.5 mb-2.5 min-h-[44px]">
                  ${item.tasks.length === 0 ? `
                    <p class="text-xs opacity-45 italic py-1">${state.lang === 'zh' ? '無排程任務' : 'No tasks'}</p>
                  ` : item.tasks.slice(0, 3).map(tk => `
                    <div class="flex items-center gap-1.5 text-xs ${tk.completed ? 'line-through opacity-50' : ''}">
                      <span class="font-mono text-[11px]">${tk.priority ? '★' : (tk.completed ? '■' : '□')}</span>
                      <span class="truncate">${escapeHtml(tk.title)}</span>
                    </div>
                  `).join('')}
                  ${item.tasks.length > 3 ? `<div class="text-[11px] opacity-55 font-mono">+${item.tasks.length - 3}</div>` : ''}
                </div>

                <!-- Footer Badges -->
                <div class="flex items-center justify-between pt-2 border-t border-current/10 text-[11px] opacity-65 font-mono">
                  <span>◇ ${item.habitsDone}/${state.habits.length}</span>
                  <span>${item.journal && item.journal.content ? (state.lang === 'zh' ? '✎ 日記' : '✎ Note') : ''}</span>
                </div>
              </div>
            `;
          }).join('')}
        </div>
      `;
    }

    // VIEW 3: MONTH CALENDAR VIEW (月曆檢視)
    // --------------------------------------------------------------------------
    function stepCalendarMonth(delta) {
      const [y, m] = state.calendarMonth.split('-').map(Number);
      const d = new Date(y, m - 1 + delta, 1);
      state.calendarMonth = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
      renderApp();
    }

    function selectCalendarDay(dateStr) {
      state.selectedDate = dateStr;
      renderApp();
    }

    function renderMonthView() {
      const [year, month] = state.calendarMonth.split('-').map(Number);
      const firstDay = new Date(year, month - 1, 1);
      const lastDay = new Date(year, month, 0);
      const startWeekday = firstDay.getDay();
      const totalDays = lastDay.getDate();

      const monthLabel = state.lang === 'zh'
        ? `${year} 年 ${month} 月`
        : firstDay.toLocaleDateString('en-US', { year: 'numeric', month: 'long' });

      const weekdays = state.lang === 'zh'
        ? ['日', '一', '二', '三', '四', '五', '六']
        : ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];

      const allMoods = getAllMoods();
      const cells = [];
      for (let i = 0; i < startWeekday; i++) cells.push(null);
      for (let d = 1; d <= totalDays; d++) {
        cells.push(`${year}-${String(month).padStart(2, '0')}-${String(d).padStart(2, '0')}`);
      }

      const selTasks = state.tasks.filter(t => t.date === state.selectedDate);
      const selMood = allMoods.find(m => m.id === state.moodLogs[state.selectedDate]);
      const selJournal = state.journals[state.selectedDate];

      return `
        <div class="duo-grid space-y-4 sm:space-y-0">
          <!-- Calendar Grid Card -->
          <section class="liquid-glass rounded-3xl p-4">
            <div class="flex items-center justify-between mb-4">
              <h2 class="text-base font-bold tracking-tight">${monthLabel}</h2>
              <div class="flex items-center gap-1.5">
                <button onclick="stepCalendarMonth(-1)" class="w-8 h-8 rounded-full liquid-pill-btn flex items-center justify-center">
                  <i data-lucide="chevron-left" class="w-4 h-4"></i>
                </button>
                <button onclick="stepCalendarMonth(1)" class="w-8 h-8 rounded-full liquid-pill-btn flex items-center justify-center">
                  <i data-lucide="chevron-right" class="w-4 h-4"></i>
                </button>
              </div>
            </div>

            <div class="grid grid-cols-7 gap-1 text-center text-[11px] font-semibold opacity-55 mb-2">
              ${weekdays.map(w => `<div>${w}</div>`).join('')}
            </div>

            <div class="grid grid-cols-7 gap-1.5">
              ${cells.map(dateKey => {
                if (!dateKey) return `<div class="h-13"></div>`;
                const dayNum = Number(dateKey.slice(-2));
                const isToday = dateKey === TODAY_KEY;
                const isSelected = dateKey === state.selectedDate;
                const tasks = state.tasks.filter(t => t.date === dateKey);
                const mood = allMoods.find(m => m.id === state.moodLogs[dateKey]);
                const hasJournal = Boolean(state.journals[dateKey] && state.journals[dateKey].content);

                return `
                  <button onclick="selectCalendarDay('${dateKey}')"
                    class="h-13 py-1.5 rounded-2xl flex flex-col items-center justify-between transition relative ${isSelected ? 'mono-solid-btn scale-105' : 'liquid-glass-subtle'} ${isToday && !isSelected ? 'ring-1.5 ring-current' : ''}">
                    <span class="text-xs font-bold leading-none font-mono">${dayNum}</span>
                    <div class="h-4 flex items-center justify-center text-xs leading-none mono-symbol">
                      ${mood ? escapeHtml(mood.emoji) : (hasJournal ? '·' : '')}
                    </div>
                    <div class="flex gap-0.5 h-1">
                      ${tasks.slice(0, 3).map(tk => `
                        <span class="w-1 h-1 rounded-full ${isSelected ? 'bg-current' : (tk.completed ? 'bg-current opacity-90' : 'bg-current opacity-35')}"></span>
                      `).join('')}
                    </div>
                  </button>
                `;
              }).join('')}
            </div>
          </section>

          <!-- Selected Date Preview Card -->
          <section class="liquid-glass rounded-3xl p-4 flex flex-col justify-between space-y-3">
            <div>
              <div class="flex items-center justify-between mb-3">
                <div>
                  <span class="text-[11px] uppercase tracking-wider opacity-55">${t('selectedDayPreview')}</span>
                  <h3 class="text-sm font-bold">${formatHumanDate(state.selectedDate)}</h3>
                </div>
                ${selMood ? `<span class="w-9 h-9 rounded-2xl mono-badge flex items-center justify-center text-base font-bold mono-symbol">${escapeHtml(selMood.emoji)}</span>` : ''}
              </div>

              <div class="space-y-2 mb-3">
                ${selTasks.length === 0 ? `
                  <p class="text-xs opacity-55 py-2">${t('noTasksToday')}</p>
                ` : selTasks.map(tk => `
                  <div class="liquid-glass-subtle rounded-2xl px-3 py-2 flex items-center justify-between text-xs">
                    <span class="${tk.completed ? 'line-through opacity-50' : 'font-medium'}">${tk.priority ? '★ ' : ''}${escapeHtml(tk.title)}</span>
                    <span class="font-mono">${tk.completed ? '■' : '□'}</span>
                  </div>
                `).join('')}
              </div>

              ${selJournal && selJournal.content ? `
                <div class="liquid-glass-subtle rounded-2xl p-3 text-xs leading-relaxed opacity-85">
                  <div class="font-semibold mb-1 opacity-65">✎ ${t('journalSection')}</div>
                  <p class="line-clamp-4 whitespace-pre-line">${escapeHtml(selJournal.content)}</p>
                </div>
              ` : ''}
            </div>

            <button onclick="selectDateAndOpenToday('${state.selectedDate}')"
              class="w-full py-2.5 rounded-2xl mono-solid-btn text-xs font-semibold flex items-center justify-center gap-1.5">
              <span>${t('openInToday')}</span>
              <i data-lucide="arrow-right" class="w-4 h-4"></i>
            </button>
          </section>
        </div>
      `;
    }

    function renderYearView() {
      const year = state.selectedYear;
      const yearGoals = state.annualGoals.filter(g => Number(g.year) === Number(year));
      const allMoods = getAllMoods();

      // Build 12-month monochrome activity cards
      const monthsHtml = Array.from({ length: 12 }, (_, mIdx) => {
        const mNum = mIdx + 1;
        const prefix = `${year}-${String(mNum).padStart(2, '0')}`;
        const mTasks = state.tasks.filter(t => t.date && t.date.startsWith(prefix));
        const mDone = mTasks.filter(t => t.completed).length;

        let activeDays = 0;
        let topMoodEmoji = '';
        Object.keys(state.moodLogs).forEach(k => {
          if (k.startsWith(prefix)) {
            activeDays++;
            const found = allMoods.find(m => m.id === state.moodLogs[k]);
            if (found) topMoodEmoji = found.emoji;
          }
        });
        Object.keys(state.journals).forEach(k => {
          if (k.startsWith(prefix) && state.journals[k].content) activeDays++;
        });

        const monthName = state.lang === 'zh'
          ? `${mNum}月`
          : new Date(year, mIdx, 1).toLocaleDateString('en-US', { month: 'short' });

        return `
          <button onclick="jumpToMonth('${prefix}')"
            class="liquid-glass-subtle rounded-2xl p-2.5 text-left hover:scale-[1.02] transition flex flex-col justify-between">
            <div class="flex items-center justify-between mb-1.5">
              <span class="text-xs font-bold">${monthName}</span>
              <span class="text-xs font-bold opacity-75 mono-symbol">${topMoodEmoji || '○'}</span>
            </div>
            <div class="text-[11px] opacity-65 font-mono">
              ${mDone}/${mTasks.length}
            </div>
            <div class="w-full h-1.5 rounded-full mt-1.5 overflow-hidden" style="background: var(--ink-track);">
              <div class="h-full rounded-full" style="width: ${Math.min(100, activeDays * 6)}%; background: var(--ink-solid);"></div>
            </div>
          </button>
        `;
      }).join('');

      return `
        <div class="duo-grid space-y-4 sm:space-y-0">
          <!-- Annual Goals Card (年度目標) -->
          <section class="liquid-glass rounded-3xl p-4">
            <div class="flex items-center justify-between mb-3">
              <div class="flex items-center gap-2">
                <span class="w-6 h-6 rounded-xl mono-badge flex items-center justify-center">
                  <i data-lucide="target" class="w-3.5 h-3.5"></i>
                </span>
                <h2 class="text-sm font-bold">${year} · ${t('annualGoalsTitle')}</h2>
              </div>
              <button onclick="openGoalModal()" class="px-2.5 py-1 rounded-full mono-solid-btn text-xs font-semibold flex items-center gap-1">
                <i data-lucide="plus" class="w-3.5 h-3.5"></i>
                <span>${t('addGoalBtn')}</span>
              </button>
            </div>

            ${yearGoals.length === 0 ? `
              <p class="text-xs opacity-60 py-6 text-center">${t('noGoalsYet')}</p>
            ` : `
              <div class="space-y-3">
                ${yearGoals.map(goal => `
                  <div class="liquid-glass-subtle rounded-2xl p-3.5 space-y-2">
                    <div class="flex items-start justify-between gap-2">
                      <div>
                        <span class="inline-block text-[10px] font-semibold px-2 py-0.5 rounded-full mono-badge mb-1">
                          ${escapeHtml(goal.category)}
                        </span>
                        <h3 class="text-sm font-bold ${goal.completed ? 'line-through opacity-55' : ''}">${escapeHtml(goal.title)}</h3>
                      </div>
                      <button onclick="openGoalModal('${goal.id}')" class="w-7 h-7 rounded-full liquid-pill-btn flex items-center justify-center shrink-0">
                        <i data-lucide="pencil" class="w-3.5 h-3.5"></i>
                      </button>
                    </div>

                    <div class="flex items-center gap-2">
                      <div class="flex-1 h-2 rounded-full overflow-hidden" style="background: var(--ink-track);">
                        <div class="h-full rounded-full transition-all duration-300"
                             style="width: ${goal.progress}%; background: var(--ink-solid);"></div>
                      </div>
                      <span class="text-xs font-bold w-9 text-right font-mono">${goal.progress}%</span>
                    </div>

                    <div class="flex items-center justify-end gap-1.5 pt-0.5">
                      <button onclick="quickIncrementGoal('${goal.id}', -10)" class="px-2 py-0.5 rounded-lg liquid-pill-btn text-[11px] font-mono font-semibold">-10%</button>
                      <button onclick="quickIncrementGoal('${goal.id}', 10)" class="px-2 py-0.5 rounded-lg liquid-pill-btn text-[11px] font-mono font-semibold">+10%</button>
                    </div>
                  </div>
                `).join('')}
              </div>
            `}
          </section>

          <!-- 12-Month Overview Matrix (年度檢視) -->
          <section class="liquid-glass rounded-3xl p-4">
            <div class="flex items-center justify-between mb-3">
              <h2 class="text-sm font-bold">${t('yearHeatmapTitle')}</h2>
              <div class="flex items-center gap-1.5">
                <button onclick="stepSelectedYear(-1)" class="w-8 h-8 rounded-full liquid-pill-btn flex items-center justify-center">
                  <i data-lucide="chevron-left" class="w-4 h-4"></i>
                </button>
                <span class="text-xs font-bold px-1 font-mono">${year}</span>
                <button onclick="stepSelectedYear(1)" class="w-8 h-8 rounded-full liquid-pill-btn flex items-center justify-center">
                  <i data-lucide="chevron-right" class="w-4 h-4"></i>
                </button>
              </div>
            </div>

            <div class="grid grid-cols-3 gap-2.5">
              ${monthsHtml}
            </div>
          </section>
        </div>
      `;
    }

      state.calendarMonth = yyyyMm;
      state.selectedDate = `${yyyyMm}-01`;
      state.activeView = 'month';
      saveState();
      renderApp();
    }
